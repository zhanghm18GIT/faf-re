# Disposable component-test build; no gameplay fix or live-scenario proof.
option(FAF_TREAD_COMPONENT_TESTS "Build production-backed tread component fixtures" OFF)
if(FAF_TREAD_COMPONENT_TESTS)
  if(NOT WIN32 OR NOT FAF_TARGET_ARCH STREQUAL "x64")
    message(FATAL_ERROR "This bounded validation configuration requires Windows x64")
  endif()
  target_compile_definitions(faf_engine PRIVATE FAF_TREAD_TEST_SEAMS=1)
  add_executable(faf_tread_component_tests EXCLUDE_FROM_ALL TreadComponentTests.cpp)
  target_compile_definitions(faf_tread_component_tests PRIVATE FAF_TREAD_TEST_SEAMS=1)
  target_link_libraries(faf_tread_component_tests PRIVATE faf_engine)
  foreach(_case IN ITEMS motion phase snapshot)
    add_test(NAME tread_${_case}_fixture COMMAND faf_tread_component_tests ${_case})
    set_tests_properties(tread_${_case}_fixture PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
  endforeach()
endif()

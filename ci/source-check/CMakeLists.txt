cmake_minimum_required(VERSION 3.25)
project(FormatterSourcePolicy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT EXISTS "${FORMATTER_CPP}")
  message(FATAL_ERROR "Generated actual-source policy regression missing")
endif()
add_executable(formatter_policy "${FORMATTER_CPP}")
target_compile_options(formatter_policy PRIVATE /W4 /EHsc)
enable_testing()
add_test(NAME formatter_policy COMMAND formatter_policy)
